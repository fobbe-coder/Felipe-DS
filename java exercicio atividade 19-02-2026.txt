
package calculomedio;


import java.util.Scanner;
public class CalculoMedio {

    public static void main(String[] args) {
      Scanner sc = new Scanner (System.in);
     System.out.println("Escreva o nome do aluno: ");
     String nomedoaluno = sc.nextLine();
     System.out.println("Escreva a primeira nota do aluno: ");
     int nota1 =  sc.nextInt();
     System.out.println("Escreva a segunda nota do aluno: ");
     int nota2=  sc.nextInt();
     System.out.println("Escreva a terceira nota do aluno: ");
     int nota3=  sc.nextInt();
     System.out.println("Escreva a quarta nota do aluno: ");
     int nota4=  sc.nextInt();
     
     double media = nota1 + nota2 + nota3 + nota4;
      media = media / 4;
     System.out.println("Aluno: " +  nomedoaluno + " Média: " + media);
        
    }
    
}
